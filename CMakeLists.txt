cmake_minimum_required(VERSION 3.24)
project(NintendoGameMusicViewer VERSION 1.3.1 LANGUAGES C CXX)
if(NOT WIN32)
 message(FATAL_ERROR "This native desktop build currently targets Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(GME_BUILD_SHARED ON CACHE BOOL "" FORCE)
set(GME_BUILD_STATIC OFF CACHE BOOL "" FORCE)
set(GME_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(GME_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GME_ENABLE_UBSAN OFF CACHE BOOL "" FORCE)
set(GME_ZLIB OFF CACHE BOOL "" FORCE)
set(GME_YM2612_EMU "GENS" CACHE STRING "" FORCE)
foreach(format AY GYM HES KSS SAP)
 set(USE_GME_${format} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(format NSF NSFE SPC GBS VGM)
 set(USE_GME_${format} ON CACHE BOOL "" FORCE)
endforeach()
FetchContent_Declare(libgme URL "${CMAKE_CURRENT_SOURCE_DIR}/third_party/game-music-emu-0.6.5.zip" URL_HASH SHA256=95444046148720dfa74ad47641dcb0232ec3e3bbd8ab5c6a013c0857da8b74bf DOWNLOAD_EXTRACT_TIMESTAMP TRUE PATCH_COMMAND ${CMAKE_COMMAND} -DGME_SOURCE_DIR=<SOURCE_DIR> -P "${CMAKE_CURRENT_SOURCE_DIR}/third_party/apply_gme_fds.cmake")
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_VIDEO OFF CACHE BOOL "" FORCE)
set(SDL_RENDER OFF CACHE BOOL "" FORCE)
set(SDL_JOYSTICK OFF CACHE BOOL "" FORCE)
set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
FetchContent_Declare(sdl URL "${CMAKE_CURRENT_SOURCE_DIR}/third_party/SDL-2.30.11.zip" URL_HASH SHA256=175eff804dface261f0ea239244da6c2f851b2b296d1ba70c42b6b710a90d6f5 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(libgme sdl)
set(GME_SOURCE_DIR "${libgme_SOURCE_DIR}")
include(third_party/apply_gme_sega.cmake)
# Audio-only GBA and DS cores; original archives and licenses are bundled.
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/zlib EXCLUDE_FROM_ALL)
add_subdirectory(third_party/viogsf EXCLUDE_FROM_ALL)
add_subdirectory(third_party/vio2sf EXCLUDE_FROM_ALL)
target_include_directories(vio2sf PUBLIC third_party/vio2sf/src/vio2sf/desmume)
target_include_directories(viogsf PUBLIC third_party/viogsf/vbam/gba)
foreach(core viogsf vio2sf)
 target_compile_definitions(${core} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
endforeach()
add_subdirectory(third_party/lazyusf EXCLUDE_FROM_ALL)
target_include_directories(lazyusf PUBLIC third_party/lazyusf)
target_compile_definitions(lazyusf PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
foreach(option USE_MPEG USE_VORBIS USE_FFMPEG USE_G7221 USE_G719 USE_ATRAC9 USE_CELT USE_SPEEX BUILD_CLI BUILD_FB2K BUILD_WINAMP BUILD_XMPLAY BUILD_AUDACIOUS BUILD_V123)
 set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(third_party/vgmstream EXCLUDE_FROM_ALL)
target_include_directories(libvgmstream PUBLIC third_party/vgmstream/src)
if(MSVC)
 target_compile_options(libvgmstream PRIVATE /utf-8)
 set_source_files_properties(third_party/vgmstream/src/base/streamfile_stdio.c TARGET_DIRECTORY libvgmstream PROPERTIES COMPILE_DEFINITIONS VGM_STDIO_UNICODE)
endif()
add_library(viewer_core STATIC src/formats/Registry.cpp src/formats/VgmBackend.cpp src/formats/GmeBackend.cpp src/formats/PsfBackend.cpp src/formats/StreamBackend.cpp src/visualizer/Envelope.cpp src/audio/SdlOutput.cpp src/player/Player.cpp src/player/FolderPlaylist.cpp)
target_include_directories(viewer_core PUBLIC src "${libgme_SOURCE_DIR}/gme" "${CMAKE_CURRENT_SOURCE_DIR}/third_party/zlib" "${CMAKE_BINARY_DIR}/third_party/zlib")
target_compile_features(viewer_core PUBLIC cxx_std_17)
target_link_libraries(viewer_core PUBLIC gme::gme SDL2::SDL2-static viogsf vio2sf zlibstatic lazyusf libvgmstream shlwapi)
target_compile_definitions(viewer_core PUBLIC SDL_MAIN_HANDLED NOMINMAX UNICODE _UNICODE)
if(MSVC)
 target_compile_options(viewer_core PRIVATE /W4 /utf-8)
endif()
add_library(viewer_desktop STATIC src/app/WindowsIntegration.cpp)
target_include_directories(viewer_desktop PUBLIC src)
target_compile_definitions(viewer_desktop PUBLIC NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0601)
target_compile_options(viewer_desktop PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_link_libraries(viewer_desktop PUBLIC advapi32 shell32 user32 comctl32)
add_executable(NintendoGameMusicViewer WIN32 src/app/main.cpp src/ui/Window.cpp src/app/app.rc src/app/app.manifest)
target_include_directories(NintendoGameMusicViewer PRIVATE src/app)
set_target_properties(NintendoGameMusicViewer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_link_libraries(NintendoGameMusicViewer PRIVATE viewer_core viewer_desktop comdlg32 shell32 gdi32 user32)
if(MSVC)
 target_link_options(NintendoGameMusicViewer PRIVATE /MAP)
endif()
target_compile_options(NintendoGameMusicViewer PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
add_custom_command(TARGET NintendoGameMusicViewer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:gme_shared> $<TARGET_FILE_DIR:NintendoGameMusicViewer>)
include(CTest)
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_executable(viewer_file_diagnostics tests/FileDiagnostics.cpp)
 target_link_libraries(viewer_file_diagnostics PRIVATE viewer_core)
 target_compile_options(viewer_file_diagnostics PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
 set_target_properties(viewer_file_diagnostics PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 add_executable(viewer_tests tests/Tests.cpp)
 set_target_properties(viewer_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 target_link_libraries(viewer_tests PRIVATE viewer_core)
 target_compile_options(viewer_tests PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
 add_custom_command(TARGET viewer_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:gme_shared> $<TARGET_FILE_DIR:viewer_tests>)
 add_test(NAME fixtures COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/make_fixtures.py" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
 set_tests_properties(fixtures PROPERTIES FIXTURES_SETUP music)
 add_test(NAME core_audio_channels COMMAND viewer_tests "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
 set_tests_properties(core_audio_channels PROPERTIES FIXTURES_REQUIRED music ENVIRONMENT "SDL_AUDIODRIVER=dummy" TIMEOUT 90)
 add_test(NAME native_ui_layout COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/ui_smoke.py" $<TARGET_FILE:NintendoGameMusicViewer> "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
 set_tests_properties(native_ui_layout PROPERTIES FIXTURES_REQUIRED music TIMEOUT 90)
 add_test(NAME native_ui_clicks COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/ui_click.py" $<TARGET_FILE:NintendoGameMusicViewer> "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
 set_tests_properties(native_ui_clicks PROPERTIES FIXTURES_REQUIRED music TIMEOUT 45)
 add_test(NAME native_folder_playlist COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/ui_playlist.py" $<TARGET_FILE:NintendoGameMusicViewer> "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
 set_tests_properties(native_folder_playlist PROPERTIES FIXTURES_REQUIRED music TIMEOUT 45)
 set_tests_properties(native_ui_clicks PROPERTIES ENVIRONMENT "NGMV_TEST_SESSION=ctest-clicks")
 set_tests_properties(native_folder_playlist PROPERTIES ENVIRONMENT "NGMV_TEST_SESSION=ctest-playlist")
 add_executable(viewer_desktop_tests tests/DesktopTests.cpp src/app/app.manifest src/app/app.rc)
 target_include_directories(viewer_desktop_tests PRIVATE src/app)
 target_link_libraries(viewer_desktop_tests PRIVATE viewer_desktop)
 target_compile_options(viewer_desktop_tests PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
 add_test(NAME windows_registration COMMAND viewer_desktop_tests)
 add_test(NAME native_default_prompt COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/ui_default_prompt.py" $<TARGET_FILE:viewer_desktop_tests>)
 set_tests_properties(native_default_prompt PROPERTIES TIMEOUT 30)
 add_test(NAME native_single_instance COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/ui_single_instance.py" $<TARGET_FILE:NintendoGameMusicViewer> "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
 set_tests_properties(native_single_instance PROPERTIES FIXTURES_REQUIRED music TIMEOUT 60)
endif()
install(TARGETS NintendoGameMusicViewer RUNTIME DESTINATION . COMPONENT Runtime)
option(NGMV_ASAN "Instrument application and GBA/DS/N64 decoders for crash diagnosis" OFF)
if(NGMV_ASAN AND MSVC)
 foreach(target gme_shared viewer_core viogsf vio2sf lazyusf NintendoGameMusicViewer viewer_tests viewer_file_diagnostics)
  if(TARGET ${target})
   target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
   get_target_property(target_type ${target} TYPE)
   if(target_type STREQUAL "EXECUTABLE" OR target_type STREQUAL "SHARED_LIBRARY")
    target_link_options(${target} PRIVATE /DEBUG /INCREMENTAL:NO)
   endif()
  endif()
 endforeach()
endif()
install(FILES $<TARGET_FILE:gme_shared> DESTINATION . COMPONENT Runtime)
install(FILES LICENSE README.md README.en.md SOURCE.md THIRD_PARTY_NOTICES.md DESTINATION . COMPONENT Runtime)
install(DIRECTORY licenses/ DESTINATION licenses COMPONENT Runtime)
